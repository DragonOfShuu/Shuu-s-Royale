#!/usr/bin/env bash
#
# Sets up the Battle Royale mod inside an empty Forge 1.20.1 MDK project.
# Run from the project root (the folder containing build.gradle).
#
set -euo pipefail

PKG_DIR="src/main/java/com/example/battleroyale"
EXAMPLE_DIR="src/main/java/com/example/examplemod"
BACKUP_DIR="examplemod_backup"

# ---------------------------------------------------------------------------
# Sanity checks
# ---------------------------------------------------------------------------
if [[ ! -f "build.gradle" ]]; then
    echo "Error: build.gradle not found. Run this script from the root of a Forge MDK project." >&2
    exit 1
fi

# ---------------------------------------------------------------------------
# Update gradle.properties so mods.toml picks up the correct mod ID
# ---------------------------------------------------------------------------
set_prop() {
    local key="$1"
    local value="$2"
    local file="gradle.properties"

    if [[ ! -f "$file" ]]; then
        echo "Warning: $file not found; set $key=$value manually." >&2
        return
    fi

    if grep -q "^${key}=" "$file"; then
        awk -v k="$key" -v v="$value" 'BEGIN { FS = "=" } $1 == k { print k "=" v; next } { print }' \
            "$file" > "$file.tmp" && mv "$file.tmp" "$file"
        echo "Set $key=$value in $file"
    else
        echo "Warning: $key not found in $file; make sure mods.toml uses modId=\"battleroyale\"." >&2
    fi
}

set_prop "mod_id" "battleroyale"
set_prop "mod_name" "Battle Royale"
set_prop "mod_group_id" "com.example.battleroyale"

# ---------------------------------------------------------------------------
# Move the MDK example mod out of the source tree (it would conflict)
# ---------------------------------------------------------------------------
if [[ -d "$EXAMPLE_DIR" ]]; then
    mkdir -p "$BACKUP_DIR"
    mv "$EXAMPLE_DIR" "$BACKUP_DIR/"
    echo "Moved MDK example mod to $BACKUP_DIR/"
fi

mkdir -p "$PKG_DIR"

package com.example.battleroyale;

import com.mojang.brigadier.arguments.IntegerArgumentType;

@Mod.EventBusSubscriber(modid = BattleRoyale.MOD_ID)
public final class ModEvents {

    private static final int MAX_BACKUP_SIZE = 4096;

    private ModEvents() {}

    @SubscribeEvent
    public static void onRegisterCommands(RegisterCommandsEvent event) {
        event.getDispatcher().register(
            Commands.literal("br")
                .requires(source -> source.hasPermission(2))
                .then(Commands.literal("start")
                    .then(Commands.argument("borderSize", IntegerArgumentType.integer(11, 59_999_968))
                    .then(Commands.argument("warTime", IntegerArgumentType.integer(1))
                    .then(Commands.argument("shrinkAmount", IntegerArgumentType.integer(1))
                    .then(Commands.argument("shrinkTime", IntegerArgumentType.integer(1))
                        .executes(ModEvents::startGame))))))
                .then(Commands.literal("stop")
                    .executes(ModEvents::stopGame))
                .then(Commands.literal("backup")
                    .then(Commands.argument("size", IntegerArgumentType.integer(11, MAX_BACKUP_SIZE))
                        .executes(ModEvents::backupArena)))
                .then(Commands.literal("reset")
                    .executes(ModEvents::resetArena))
        );
    }

    private static boolean requireOverworld(CommandSourceStack source) {
        if (source.getLevel() != source.getServer().overworld()) {
            source.sendFailure(Component.literal("This command must be run in the Overworld."));
            return false;
        }
        return true;
    }

    private static int startGame(CommandContext<CommandSourceStack> ctx) {
        CommandSourceStack source = ctx.getSource();
        if (!requireOverworld(source)) return 0;

        if (BattleRoyaleGame.isRunning()) {
            source.sendFailure(Component.literal("A game is already running."));
            return 0;
        }
        if (ArenaSnapshot.isBusy()) {
            source.sendFailure(Component.literal("Wait for the current backup or restore to finish."));
            return 0;
        }

        Vec3 pos = source.getPosition();
        BattleRoyaleGame.start(
            source.getServer(),
            pos.x, pos.z,
            IntegerArgumentType.getInteger(ctx, "borderSize"),
            IntegerArgumentType.getInteger(ctx, "warTime"),
            IntegerArgumentType.getInteger(ctx, "shrinkAmount"),
            IntegerArgumentType.getInteger(ctx, "shrinkTime")
        );
        return 1;
    }

    private static int stopGame(CommandContext<CommandSourceStack> ctx) {
        if (!BattleRoyaleGame.isActive()) {
            ctx.getSource().sendFailure(Component.literal("No game is running."));
            return 0;
        }
        BattleRoyaleGame.stop(ctx.getSource().getServer());
        return 1;
    }

    private static int backupArena(CommandContext<CommandSourceStack> ctx) {
        CommandSourceStack source = ctx.getSource();
        if (!requireOverworld(source)) return 0;

        if (BattleRoyaleGame.isRunning()) {
            source.sendFailure(Component.literal("Stop the current game before making a backup."));
            return 0;
        }
        if (ArenaSnapshot.isBusy()) {
            source.sendFailure(Component.literal("A backup or restore is already in progress."));
            return 0;
        }

        Vec3 pos = source.getPosition();
        ArenaSnapshot.beginCapture(source.getServer(), pos.x, pos.z,
                IntegerArgumentType.getInteger(ctx, "size"));
        return 1;
    }

    private static int resetArena(CommandContext<CommandSourceStack> ctx) {
        CommandSourceStack source = ctx.getSource();

        if (ArenaSnapshot.isBusy()) {
            source.sendFailure(Component.literal("A backup or restore is already in progress."));
            return 0;
        }
        if (!ArenaSnapshot.hasBackup()) {
            source.sendFailure(Component.literal("No arena backup exists. Use /br backup <size> first."));
            return 0;
        }

        if (BattleRoyaleGame.isActive()) {
            BattleRoyaleGame.stop(source.getServer());
        }
        ArenaSnapshot.beginRestore(source.getServer());
        return 1;
    }

    @SubscribeEvent
    public static void onServerTick(TickEvent.ServerTickEvent event) {
        if (event.phase == TickEvent.Phase.END) {
            MinecraftServer server = event.getServer();
            ArenaSnapshot.tick(server);
            BattleRoyaleGame.tick(server);
            PlayerDrop.tick(server);
        }
    }

    @SubscribeEvent
    public static void onServerStarted(ServerStartedEvent event) {
        ArenaSnapshot.loadFromDisk(event.getServer());
    }

    @SubscribeEvent
    public static void onServerStopping(ServerStoppingEvent event) {
        // Don't let the server shut down halfway through writing a backup
        ArenaSnapshot.awaitPendingIo();
    }

    @SubscribeEvent
    public static void onServerStopped(ServerStoppedEvent event) {
        // Static state survives between singleplayer worlds in the same session, so clear it.
        BattleRoyaleGame.reset();
        ArenaSnapshot.clear();
        PlayerDrop.clear();
    }

    @SubscribeEvent
    public static void onPlayerLogin(PlayerEvent.PlayerLoggedInEvent event) {
        if (event.getEntity() instanceof ServerPlayer player) {
            PlayerDrop.onLogin(player);
        }
    }

    @SubscribeEvent
    public static void onPlayerLogout(PlayerEvent.PlayerLoggedOutEvent event) {
        if (event.getEntity() instanceof ServerPlayer player) {
            BattleRoyaleGame.onPlayerLeave(player);
            PlayerDrop.onLogout(player);
        }
    }

    @SubscribeEvent
    public static void onPlayerRespawn(PlayerEvent.PlayerRespawnEvent event) {
        if (event.getEntity() instanceof ServerPlayer player) {
            PlayerDrop.onRespawn(player);
        }
    }

    @SubscribeEvent
    public static void onLivingFall(LivingFallEvent event) {
        if (event.getEntity() instanceof ServerPlayer player && PlayerDrop.isDropping(player)) {
            event.setCanceled(true);
        }
    }

    @SubscribeEvent
    public static void onLivingAttack(LivingAttackEvent event) {
        if (event.getEntity() instanceof ServerPlayer player && PlayerDrop.isDropping(player)
                && (event.getSource().is(DamageTypes.FALL) || event.getSource().is(DamageTypes.FLY_INTO_WALL))) {
            event.setCanceled(true);
        }
    }

    @SubscribeEvent
    public static void onLivingDeath(LivingDeathEvent event) {
        if (event.getEntity() instanceof ServerPlayer player) {
            PlayerDrop.onDeath(player);
        }
    }
}

package com.example.battleroyale;

import net.minecraft.ChatFormatting;
import net.minecraft.network.chat.Component;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.level.ServerBossEvent;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.util.Mth;
import net.minecraft.world.BossEvent;
import net.minecraft.world.level.border.WorldBorder;

public final class BattleRoyaleGame {

    private static final double MIN_SIZE = 10.0;
    private static final int TICKS_PER_SECOND = 20;
    private static final long MILLIS_PER_TICK = 50L;

    private enum Phase { IDLE, WAR, SHRINKING, FINISHED }

    private static final ServerBossEvent BAR = new ServerBossEvent(
            Component.literal("Battle Royale"),
            BossEvent.BossBarColor.YELLOW,
            BossEvent.BossBarOverlay.PROGRESS);

    private static Phase phase = Phase.IDLE;
    private static int warTicks;
    private static int shrinkTicks;
    private static int shrinkAmount;
    private static int timer;
    private static double targetSize;

    private BattleRoyaleGame() {}

    /** True while the border cycle is active (war or shrinking). */
    public static boolean isRunning() {
        return phase == Phase.WAR || phase == Phase.SHRINKING;
    }

    /** True whenever a game exists, including after the border has finished shrinking. */
    public static boolean isActive() {
        return phase != Phase.IDLE;
    }

    public static void start(MinecraftServer server, double centerX, double centerZ,
                             int borderSize, int warTime, int shrinkAmount, int shrinkTime) {
        ServerLevel level = server.overworld();
        WorldBorder border = level.getWorldBorder();
        border.setCenter(centerX, centerZ);
        border.setSize(borderSize);
        border.setWarningBlocks(5);

        BattleRoyaleGame.warTicks = warTime * TICKS_PER_SECOND;
        BattleRoyaleGame.shrinkTicks = shrinkTime * TICKS_PER_SECOND;
        BattleRoyaleGame.shrinkAmount = shrinkAmount;
        BattleRoyaleGame.targetSize = borderSize;

        phase = Phase.WAR;
        timer = warTicks;

        // Drop participants into the arena and hand out starting gear
        for (ServerPlayer player : server.getPlayerList().getPlayers()) {
            if (player.isSpectator() || player.isCreative()) continue;
            PlayerDrop.drop(level, player, centerX, centerZ, borderSize);
            Loadout.giveLoadout(player);
        }

        updateBar();
        addAllPlayers(server);

        broadcast(server, "The battle has begun! Border: " + borderSize
                + " blocks. First shrink in " + warTime + " seconds.");
    }

    public static void stop(MinecraftServer server) {
        WorldBorder border = server.overworld().getWorldBorder();
        border.setSize(border.getSize()); // Freeze any shrink in progress
        reset();
        broadcast(server, "The game has been stopped.");
    }

    /** Clears game state and hides the boss bar without touching the world. */
    public static void reset() {
        phase = Phase.IDLE;
        BAR.removeAllPlayers();
    }

    public static void onPlayerLeave(ServerPlayer player) {
        BAR.removePlayer(player);
    }

    public static void tick(MinecraftServer server) {
        if (phase == Phase.IDLE) return;

        // Make sure players who joined mid-game can see the bar
        if (server.getTickCount() % TICKS_PER_SECOND == 0) {
            addAllPlayers(server);
        }

        if (phase == Phase.FINISHED) return;

        if (--timer <= 0) {
            WorldBorder border = server.overworld().getWorldBorder();
            if (phase == Phase.WAR) {
                beginShrink(server, border);
            } else {
                endShrink(server);
            }
        }

        updateBar();
    }

    private static void beginShrink(MinecraftServer server, WorldBorder border) {
        double current = targetSize;
        targetSize = Math.max(MIN_SIZE, current - shrinkAmount);

        border.lerpSizeBetween(current, targetSize, shrinkTicks * MILLIS_PER_TICK);

        phase = Phase.SHRINKING;
        timer = shrinkTicks;

        broadcast(server, "The border is shrinking to " + (int) targetSize + " blocks!");
    }

    private static void endShrink(MinecraftServer server) {
        if (targetSize <= MIN_SIZE) {
            phase = Phase.FINISHED;
            broadcast(server, "The border has reached its final size of " + (int) MIN_SIZE + " blocks!");
            return;
        }

        phase = Phase.WAR;
        timer = warTicks;
        broadcast(server, "The border has stopped. Next shrink in "
                + (warTicks / TICKS_PER_SECOND) + " seconds.");
    }

    private static void updateBar() {
        switch (phase) {
            case WAR -> {
                BAR.setColor(BossEvent.BossBarColor.YELLOW);
                BAR.setName(Component.literal("Border shrinks in " + formatTime(timer)));
                BAR.setProgress(Mth.clamp((float) timer / warTicks, 0.0F, 1.0F));
            }
            case SHRINKING -> {
                BAR.setColor(BossEvent.BossBarColor.RED);
                BAR.setName(Component.literal("Border shrinking to " + (int) targetSize
                        + " blocks: " + formatTime(timer)));
                BAR.setProgress(Mth.clamp((float) timer / shrinkTicks, 0.0F, 1.0F));
            }
            case FINISHED -> {
                BAR.setColor(BossEvent.BossBarColor.PURPLE);
                BAR.setName(Component.literal("Final border: " + (int) MIN_SIZE + " blocks"));
                BAR.setProgress(1.0F);
            }
            default -> { }
        }
    }

    private static void addAllPlayers(MinecraftServer server) {
        for (ServerPlayer player : server.getPlayerList().getPlayers()) {
            BAR.addPlayer(player);
        }
    }

    private static String formatTime(int ticks) {
        int seconds = (ticks + TICKS_PER_SECOND - 1) / TICKS_PER_SECOND;
        return String.format("%d:%02d", seconds / 60, seconds % 60);
    }

    public static void broadcast(MinecraftServer server, String message) {
        server.getPlayerList().broadcastSystemMessage(
                Component.literal("[BR] ").withStyle(ChatFormatting.GOLD)
                        .append(Component.literal(message).withStyle(ChatFormatting.WHITE)),
                false);
    }
}

import java.util.concurrent.CompletableFuture;
import java.util.stream.Stream;

/**
 * Captures the blocks of an arena area, saves them to disk, and restores them later.
 * Capture and restore are spread across ticks; disk I/O runs on a background thread.
 */
public final class ArenaSnapshot {

    private static final Logger LOGGER = LogUtils.getLogger();

    private enum State { EMPTY, CAPTURING, SAVING, LOADING, READY, RESTORING }

    private static final int CAPTURE_CHUNKS_PER_TICK = 4;
    private static final int RESTORE_CHUNKS_PER_TICK = 2;

    // Update clients, skip neighbor shape updates (prevents cascading physics), never drop items.
    private static final int RESTORE_FLAGS =
            Block.UPDATE_CLIENTS | Block.UPDATE_KNOWN_SHAPE | Block.UPDATE_SUPPRESS_DROPS;

    private static final String BACKUP_FOLDER = "battleroyale_backup";
    private static final String META_FILE = "meta.dat";

    // Same codec vanilla uses to store chunk sections in region files.
    private static final Codec<PalettedContainer<BlockState>> SECTION_CODEC = PalettedContainer.codecRW(
            Block.BLOCK_STATE_REGISTRY, BlockState.CODEC,
            PalettedContainer.Strategy.SECTION_STATES, Blocks.AIR.defaultBlockState());

    private record ChunkData(PalettedContainer<BlockState>[] sections,
                             Map<BlockPos, CompoundTag> blockEntities) {}

    private record Backup(double centerX, double centerZ, int size, Map<Long, ChunkData> chunks) {}

    private static final Map<Long, ChunkData> SAVED = new HashMap<>();
    private static final ArrayDeque<ChunkPos> QUEUE = new ArrayDeque<>();

    private static State state = State.EMPTY;
    private static int totalChunks;
    private static int lastReportedStep;
    private static double centerX;
    private static double centerZ;
    private static int size;
    private static CompletableFuture<?> pendingIo = CompletableFuture.completedFuture(null);

    private ArenaSnapshot() {}

    public static boolean isBusy() {
        return state == State.CAPTURING || state == State.SAVING
                || state == State.LOADING || state == State.RESTORING;
    }

    public static boolean hasBackup() {
        return state == State.READY;
    }

    public static void clear() {
        SAVED.clear();
        QUEUE.clear();
        state = State.EMPTY;
    }

    // ------------------------------------------------------------------
    // Capture / restore jobs
    // ------------------------------------------------------------------

    public static void beginCapture(MinecraftServer server, double cx, double cz, int arenaSize) {
        clear();
        centerX = cx;
        centerZ = cz;
        size = arenaSize;

        double half = arenaSize / 2.0;
        int minChunkX = SectionPos.blockToSectionCoord(Mth.floor(cx - half));
        int maxChunkX = SectionPos.blockToSectionCoord(Mth.floor(cx + half));
        int minChunkZ = SectionPos.blockToSectionCoord(Mth.floor(cz - half));
        int maxChunkZ = SectionPos.blockToSectionCoord(Mth.floor(cz + half));

        for (int x = minChunkX; x <= maxChunkX; x++) {
            for (int z = minChunkZ; z <= maxChunkZ; z++) {
                QUEUE.add(new ChunkPos(x, z));
            }
        }

        startJob(State.CAPTURING);
        BattleRoyaleGame.broadcast(server, "Backing up arena (" + totalChunks + " chunks)...");
    }

    public static void beginRestore(MinecraftServer server) {
        QUEUE.clear();
        for (long key : SAVED.keySet()) {
            QUEUE.add(new ChunkPos(key));
        }

        startJob(State.RESTORING);
        BattleRoyaleGame.broadcast(server, "Restoring arena (" + totalChunks + " chunks)...");
    }

    private static void startJob(State newState) {
        totalChunks = QUEUE.size();
        lastReportedStep = 0;
        state = newState;
    }

    public static void tick(MinecraftServer server) {
        if (state != State.CAPTURING && state != State.RESTORING) return;

        ServerLevel level = server.overworld();
        int budget = state == State.CAPTURING ? CAPTURE_CHUNKS_PER_TICK : RESTORE_CHUNKS_PER_TICK;

        for (int i = 0; i < budget && !QUEUE.isEmpty(); i++) {
            ChunkPos pos = QUEUE.poll();
            if (state == State.CAPTURING) {
                captureChunk(level, pos);
            } else {
                restoreChunk(level, pos);
            }
        }

        reportProgress(server);

        if (QUEUE.isEmpty()) {
            finish(server, level);
        }
    }

    private static void captureChunk(ServerLevel level, ChunkPos pos) {
        LevelChunk chunk = level.getChunk(pos.x, pos.z);
        LevelChunkSection[] sections = chunk.getSections();

        @SuppressWarnings("unchecked")
        PalettedContainer<BlockState>[] copies = new PalettedContainer[sections.length];
        for (int i = 0; i < sections.length; i++) {
            copies[i] = sections[i].getStates().copy();
        }

        Map<BlockPos, CompoundTag> blockEntities = new HashMap<>();
        for (BlockEntity blockEntity : chunk.getBlockEntities().values()) {
            blockEntities.put(blockEntity.getBlockPos().immutable(), blockEntity.saveWithFullMetadata());
        }

        SAVED.put(pos.toLong(), new ChunkData(copies, blockEntities));
    }

    private static void restoreChunk(ServerLevel level, ChunkPos pos) {
        ChunkData data = SAVED.get(pos.toLong());
        if (data == null) return;

        LevelChunk chunk = level.getChunk(pos.x, pos.z);
        LevelChunkSection[] sections = chunk.getSections();
        BlockPos.MutableBlockPos cursor = new BlockPos.MutableBlockPos();
        int baseX = pos.getMinBlockX();
        int baseZ = pos.getMinBlockZ();
        int sectionCount = Math.min(sections.length, data.sections().length);

        // Restore block states, only touching blocks that actually changed
        for (int i = 0; i < sectionCount; i++) {
            PalettedContainer<BlockState> saved = data.sections()[i];
            if (saved == null) continue; // Section failed to load from disk; leave it alone

            LevelChunkSection current = sections[i];
            int baseY = SectionPos.sectionToBlockCoord(chunk.getSectionYFromSectionIndex(i));

            for (int y = 0; y < 16; y++) {
                for (int z = 0; z < 16; z++) {
                    for (int x = 0; x < 16; x++) {
                        BlockState wanted = saved.get(x, y, z);
                        if (current.getBlockState(x, y, z) != wanted) {
                            cursor.set(baseX + x, baseY + y, baseZ + z);
                            level.setBlock(cursor, wanted, RESTORE_FLAGS);
                        }
                    }
                }
            }
        }

        // Restore block entity data (chest contents, sign text, etc.)
        for (Map.Entry<BlockPos, CompoundTag> entry : data.blockEntities().entrySet()) {
            BlockPos bePos = entry.getKey();
            BlockEntity blockEntity = level.getBlockEntity(bePos);
            if (blockEntity != null) {
                blockEntity.load(entry.getValue().copy());
                blockEntity.setChanged();
                BlockState blockState = level.getBlockState(bePos);
                level.sendBlockUpdated(bePos, blockState, blockState, Block.UPDATE_CLIENTS);
            }
        }
    }

    private static void finish(MinecraftServer server, ServerLevel level) {
        if (state == State.CAPTURING) {
            saveAsync(server);
            return;
        }

        clearLeftoverEntities(level);

        WorldBorder border = level.getWorldBorder();
        border.setCenter(centerX, centerZ);
        border.setSize(size);

        state = State.READY;
        BattleRoyaleGame.broadcast(server, "Arena restored.");
    }

    private static void clearLeftoverEntities(ServerLevel level) {
        double half = size / 2.0 + 16;
        AABB area = new AABB(
                centerX - half, level.getMinBuildHeight(), centerZ - half,
                centerX + half, level.getMaxBuildHeight(), centerZ + half);

        List<Entity> leftovers = level.getEntities((Entity) null, area, entity ->
                entity instanceof ItemEntity
                        || entity instanceof PrimedTnt
                        || entity instanceof FallingBlockEntity
                        || entity instanceof ExperienceOrb
                        || entity instanceof Projectile);

        leftovers.forEach(Entity::discard);
    }

    private static void reportProgress(MinecraftServer server) {
        if (totalChunks == 0) return;
        int done = totalChunks - QUEUE.size();
        int step = done * 10 / totalChunks; // Report every 10%
        if (step > lastReportedStep && step < 10) {
            lastReportedStep = step;
            String verb = state == State.CAPTURING ? "Backup" : "Restore";
            BattleRoyaleGame.broadcast(server, verb + " progress: " + (step * 10) + "%");
        }
    }

    // ------------------------------------------------------------------
    // Disk persistence
    // ------------------------------------------------------------------

    private static Path backupDir(MinecraftServer server) {
        return server.getWorldPath(LevelResource.ROOT).resolve(BACKUP_FOLDER).normalize();
    }

    private static void saveAsync(MinecraftServer server) {
        state = State.SAVING;
        BattleRoyaleGame.broadcast(server, "Arena captured. Saving backup to disk...");

        Backup backup = new Backup(centerX, centerZ, size, new HashMap<>(SAVED));
        Path dir = backupDir(server);

        pendingIo = CompletableFuture.runAsync(() -> writeToDisk(dir, backup), Util.ioPool())
                .whenComplete((ignored, error) -> server.execute(() -> {
                    state = State.READY;
                    if (error != null) {
                        LOGGER.error("Failed to save arena backup", error);
                        BattleRoyaleGame.broadcast(server,
                                "Saving the backup failed; it is only kept in memory. Check the server log.");
                    } else {
                        BattleRoyaleGame.broadcast(server, "Arena backup saved.");
                    }
                }));
    }

    /** Called when the server starts. Loads an existing backup in the background. */
    public static void loadFromDisk(MinecraftServer server) {
        Path dir = backupDir(server);
        if (!Files.exists(dir.resolve(META_FILE))) return;

        state = State.LOADING;
        LOGGER.info("Loading arena backup from {}", dir);

        pendingIo = CompletableFuture.supplyAsync(() -> readFromDisk(dir), Util.ioPool())
                .whenComplete((backup, error) -> server.execute(() -> {
                    if (error != null || backup == null) {
                        LOGGER.error("Failed to load arena backup", error);
                        state = State.EMPTY;
                        return;
                    }
                    SAVED.clear();
                    SAVED.putAll(backup.chunks());
                    centerX = backup.centerX();
                    centerZ = backup.centerZ();
                    size = backup.size();
                    state = State.READY;
                    LOGGER.info("Loaded arena backup: {} chunks, size {}", backup.chunks().size(), backup.size());
                }));
    }

    /** Blocks until any in-progress save or load finishes. Call while the server is stopping. */
    public static void awaitPendingIo() {
        try {
            pendingIo.join();
        } catch (Exception e) {
            LOGGER.error("Arena backup I/O failed during shutdown", e);
        }
    }

    private static void writeToDisk(Path dir, Backup backup) {
        try {
            Files.createDirectories(dir);

            // Remove the metadata first so a half-written backup is never loaded
            Files.deleteIfExists(dir.resolve(META_FILE));
            try (Stream<Path> files = Files.list(dir)) {
                for (Path file : files.toList()) {
                    if (file.getFileName().toString().endsWith(".dat")) {
                        Files.delete(file);
                    }
                }
            }

            // Group chunks into one file per 32x32-chunk region
            Map<Long, ListTag> regions = new HashMap<>();
            for (Map.Entry<Long, ChunkData> entry : backup.chunks().entrySet()) {
                ChunkPos pos = new ChunkPos(entry.getKey());
                long regionKey = ChunkPos.asLong(pos.getRegionX(), pos.getRegionZ());
                regions.computeIfAbsent(regionKey, key -> new ListTag())
                        .add(encodeChunk(pos, entry.getValue()));
            }

            for (Map.Entry<Long, ListTag> entry : regions.entrySet()) {
                ChunkPos region = new ChunkPos(entry.getKey());
                CompoundTag root = new CompoundTag();
                root.put("Chunks", entry.getValue());
                writeTag(dir.resolve("r." + region.x + "." + region.z + ".dat"), root);
            }

            CompoundTag meta = new CompoundTag();
            meta.putInt("DataVersion", currentDataVersion());
            meta.putDouble("CenterX", backup.centerX());
            meta.putDouble("CenterZ", backup.centerZ());
            meta.putInt("Size", backup.size());
            meta.putInt("ChunkCount", backup.chunks().size());
            writeTag(dir.resolve(META_FILE), meta);
        } catch (IOException e) {
            throw new UncheckedIOException(e);
        }
    }

    private static Backup readFromDisk(Path dir) {
        try {
            CompoundTag meta = readTag(dir.resolve(META_FILE));
            int dataVersion = meta.getInt("DataVersion");
            if (dataVersion != currentDataVersion()) {
                LOGGER.warn("Arena backup was made with data version {} (current {}). Some blocks may not restore correctly.",
                        dataVersion, currentDataVersion());
            }

            Map<Long, ChunkData> chunks = new HashMap<>();
            try (Stream<Path> files = Files.list(dir)) {
                for (Path file : files.toList()) {
                    String name = file.getFileName().toString();
                    if (!name.startsWith("r.") || !name.endsWith(".dat")) continue;

                    ListTag list = readTag(file).getList("Chunks", Tag.TAG_COMPOUND);
                    for (int i = 0; i < list.size(); i++) {
                        CompoundTag chunkTag = list.getCompound(i);
                        ChunkPos pos = new ChunkPos(chunkTag.getInt("X"), chunkTag.getInt("Z"));
                        chunks.put(pos.toLong(), decodeChunk(chunkTag));
                    }
                }
            }

            int expected = meta.getInt("ChunkCount");
            if (chunks.size() != expected) {
                LOGGER.warn("Arena backup expected {} chunks but found {}", expected, chunks.size());
            }

            return new Backup(meta.getDouble("CenterX"), meta.getDouble("CenterZ"), meta.getInt("Size"), chunks);
        } catch (IOException e) {
            throw new UncheckedIOException(e);
        }
    }

    private static CompoundTag encodeChunk(ChunkPos pos, ChunkData data) {
        CompoundTag tag = new CompoundTag();
        tag.putInt("X", pos.x);
        tag.putInt("Z", pos.z);

        ListTag sections = new ListTag();
        for (PalettedContainer<BlockState> section : data.sections()) {
            sections.add(SECTION_CODEC.encodeStart(NbtOps.INSTANCE, section).getOrThrow(false, LOGGER::error));
        }
        tag.put("Sections", sections);

        ListTag blockEntities = new ListTag();
        for (CompoundTag blockEntity : data.blockEntities().values()) {
            blockEntities.add(blockEntity.copy());
        }
        tag.put("BlockEntities", blockEntities);

        return tag;
    }

    private static ChunkData decodeChunk(CompoundTag tag) {
        ListTag sectionTags = tag.getList("Sections", Tag.TAG_COMPOUND);

        @SuppressWarnings("unchecked")
        PalettedContainer<BlockState>[] sections = new PalettedContainer[sectionTags.size()];
        for (int i = 0; i < sectionTags.size(); i++) {
            sections[i] = SECTION_CODEC.parse(NbtOps.INSTANCE, sectionTags.getCompound(i))
                    .resultOrPartial(LOGGER::error)
                    .orElse(null);
        }

        Map<BlockPos, CompoundTag> blockEntities = new HashMap<>();
        ListTag beTags = tag.getList("BlockEntities", Tag.TAG_COMPOUND);
        for (int i = 0; i < beTags.size(); i++) {
            CompoundTag beTag = beTags.getCompound(i);
            blockEntities.put(BlockEntity.getPosFromTag(beTag), beTag);
        }

        return new ChunkData(sections, blockEntities);
    }

    private static void writeTag(Path path, CompoundTag tag) throws IOException {
        try (OutputStream out = Files.newOutputStream(path)) {
            NbtIo.writeCompressed(tag, out);
        }
    }

    private static CompoundTag readTag(Path path) throws IOException {
        try (InputStream in = Files.newInputStream(path)) {
            return NbtIo.readCompressed(in);
        }
    }

    private static int currentDataVersion() {
        return SharedConstants.getCurrentVersion().getDataVersion().getVersion();
    }
}

package com.example.battleroyale;

import net.minecraft.ChatFormatting;
import net.minecraft.nbt.CompoundTag;
import net.minecraft.network.chat.Component;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.util.RandomSource;
import net.minecraft.world.entity.EquipmentSlot;
import net.minecraft.world.entity.player.Inventory;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.item.Items;
import net.minecraft.world.item.enchantment.Enchantments;

import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.UUID;

/**
 * Handles dropping players into the arena from the sky with a temporary elytra
 * and fall damage immunity until they land.
 */
public final class PlayerDrop {

    /** Absolute Y level players are teleported to. */
    private static final double DROP_HEIGHT = 500.0;
    /** Players spawn within this fraction of the border radius, keeping them away from the edge. */
    private static final double SPREAD_FRACTION = 0.8;
    /** Ticks to wait after teleporting before checking for landing. */
    private static final int GRACE_TICKS = 20;

    private static final String DATA_KEY = "BattleRoyaleDrop";
    private static final String ELYTRA_MARKER = "BattleRoyaleDropElytra";

    /** Online players currently dropping, mapped to ticks spent in the drop. */
    private static final Map<UUID, Integer> DROPPING = new HashMap<>();

    private PlayerDrop() {}

    public static boolean isDropping(ServerPlayer player) {
        return DROPPING.containsKey(player.getUUID());
    }

    public static void clear() {
        DROPPING.clear();
    }

    public static void drop(ServerLevel level, ServerPlayer player,
                            double centerX, double centerZ, double borderSize) {
        if (isDropping(player)) {
            finishDrop(player);
        }

        RandomSource random = level.getRandom();
        double radius = borderSize / 2.0 * SPREAD_FRACTION;
        double x = centerX + (random.nextDouble() * 2.0 - 1.0) * radius;
        double z = centerZ + (random.nextDouble() * 2.0 - 1.0) * radius;

        player.stopRiding();
        player.teleportTo(level, x, DROP_HEIGHT, z, player.getYRot(), player.getXRot());
        player.fallDistance = 0.0F;

        // Remember the chest slot item so it can be returned on landing.
        // Stored in persistent player data so it survives logouts and restarts.
        CompoundTag dropData = new CompoundTag();
        dropData.put("PreviousChest", player.getItemBySlot(EquipmentSlot.CHEST).copy().save(new CompoundTag()));
        player.getPersistentData().put(DATA_KEY, dropData);

        player.setItemSlot(EquipmentSlot.CHEST, createDropElytra());
        DROPPING.put(player.getUUID(), 0);

        player.sendSystemMessage(Component.literal("Press jump while falling to open your elytra!")
                .withStyle(ChatFormatting.AQUA));
    }

    public static void tick(MinecraftServer server) {
        if (DROPPING.isEmpty()) return;

        for (UUID id : List.copyOf(DROPPING.keySet())) {
            ServerPlayer player = server.getPlayerList().getPlayer(id);
            if (player == null || !player.isAlive()) continue;

            int ticks = DROPPING.merge(id, 1, Integer::sum);
            if (ticks < GRACE_TICKS) continue;

            if (player.onGround() || player.isInWater() || player.isInLava()
                    || player.isPassenger() || player.isSpectator()) {
                finishDrop(player);
            }
        }
    }

    /** Removes the drop elytra, returns the previous chest item, and ends fall immunity. */
    public static void finishDrop(ServerPlayer player) {
        DROPPING.remove(player.getUUID());
        removeDropElytras(player);

        CompoundTag persistent = player.getPersistentData();
        if (persistent.contains(DATA_KEY)) {
            ItemStack previous = readPreviousChest(persistent);
            persistent.remove(DATA_KEY);

            if (!previous.isEmpty()) {
                if (player.getItemBySlot(EquipmentSlot.CHEST).isEmpty()) {
                    player.setItemSlot(EquipmentSlot.CHEST, previous);
                } else if (!player.getInventory().add(previous)) {
                    player.drop(previous, false);
                }
            }
        }

        player.fallDistance = 0.0F;
    }

    public static void onLogin(ServerPlayer player) {
        if (player.getPersistentData().contains(DATA_KEY)) {
            // Player logged out mid-drop; resume protection
            DROPPING.put(player.getUUID(), 0);
        } else {
            removeDropElytras(player);
        }
    }

    public static void onLogout(ServerPlayer player) {
        // Persistent data keeps the drop state; it resumes on next login
        DROPPING.remove(player.getUUID());
    }

    public static void onRespawn(ServerPlayer player) {
        if (!isDropping(player)) {
            removeDropElytras(player); // Covers keepInventory
        }
    }

    public static void onDeath(ServerPlayer player) {
        CompoundTag persistent = player.getPersistentData();
        if (!persistent.contains(DATA_KEY)) return;

        ItemStack previous = readPreviousChest(persistent);
        persistent.remove(DATA_KEY);
        DROPPING.remove(player.getUUID());

        if (!previous.isEmpty()) {
            player.drop(previous, true, false);
        }
    }

    private static ItemStack readPreviousChest(CompoundTag persistent) {
        return ItemStack.of(persistent.getCompound(DATA_KEY).getCompound("PreviousChest"));
    }

    private static ItemStack createDropElytra() {
        ItemStack elytra = new ItemStack(Items.ELYTRA);
        elytra.enchant(Enchantments.BINDING_CURSE, 1);
        elytra.enchant(Enchantments.VANISHING_CURSE, 1);
        elytra.getOrCreateTag().putBoolean("Unbreakable", true);
        elytra.getOrCreateTag().putBoolean(ELYTRA_MARKER, true);
        elytra.setHoverName(Component.literal("Drop Elytra").withStyle(ChatFormatting.AQUA));
        return elytra;
    }

    private static boolean isDropElytra(ItemStack stack) {
        return stack.is(Items.ELYTRA) && stack.hasTag() && stack.getTag().getBoolean(ELYTRA_MARKER);
    }

    private static void removeDropElytras(ServerPlayer player) {
        Inventory inventory = player.getInventory();
        for (int i = 0; i < inventory.getContainerSize(); i++) {
            if (isDropElytra(inventory.getItem(i))) {
                inventory.setItem(i, ItemStack.EMPTY);
            }
        }
        if (isDropElytra(player.containerMenu.getCarried())) {
            player.containerMenu.setCarried(ItemStack.EMPTY);
        }
    }
}

    /**
     * Gives an item by registry ID (e.g. "minecraft:arrow").
     * Looks the item up at runtime, so the gun mod is not a compile-time dependency.
     */
    private static void give(ServerPlayer player, String itemId, int count) {
        ResourceLocation id = ResourceLocation.tryParse(itemId);
        Item item = id == null ? null : ForgeRegistries.ITEMS.getValue(id);

        if (item == null || item == Items.AIR) {
            LOGGER.warn("Loadout item '{}' does not exist; skipping", itemId);
            return;
        }

        ItemStack stack = new ItemStack(item, count);
        if (!player.getInventory().add(stack)) {
            player.drop(stack, false); // Inventory full; drop what's left at their feet
        }
    }
